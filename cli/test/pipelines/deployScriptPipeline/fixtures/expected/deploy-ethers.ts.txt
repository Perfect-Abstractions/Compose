import { network } from "hardhat";

const { ethers, networkName } = await network.create();

async function deploy() {
  console.log(`Deploying Diamond to ${networkName}...`);

  const facets: string[] = [];

  // Base facet generation.
  const localFacet = await ethers.deployContract("contracts/nested/counter/LocalFacet.sol:LocalFacet");
  await localFacet.waitForDeployment();
  console.log("LocalFacet:", await localFacet.getAddress());
  facets.push(await localFacet.getAddress());

  const packageFacet = await ethers.deployContract("sample-facets/PackageFacet.sol:PackageFacet");
  await packageFacet.waitForDeployment();
  console.log("PackageFacet:", await packageFacet.getAddress());
  facets.push(await packageFacet.getAddress());

  // Library facet generation.


  // Define diamond proxy.
  const diamond = await ethers.deployContract("contracts/proxy/Proxy.sol:CustomDiamond", [facets]);
  await diamond.waitForDeployment();

  console.log("Diamond:", await diamond.getAddress());
}

deploy().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
